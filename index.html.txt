  <script>
    let globalData = {};
    let chartSCurveObj, chartEVMObj, chartCashFlowObj;
    let modalObj;

    document.addEventListener('DOMContentLoaded', function() {
      modalObj = new bootstrap.Modal(document.getElementById('transModal'));
    });

    // 1. HANDLER LOGIN
    function handleLogin(e) {
      e.preventDefault();
      const u = document.getElementById('username').value;
      const p = document.getElementById('password').value;

      google.script.run.withSuccessHandler(function(res) {
        if (res.success) {
          document.getElementById('loginView').classList.add('d-none');
          document.getElementById('dashboardView').classList.remove('d-none');
          document.getElementById('userDisplayName').innerHTML = `<i class="bi bi-person-circle me-1"></i> ${res.user.name} (${res.user.role})`;
          loadDashboard();
        } else {
          const alert = document.getElementById('loginAlert');
          alert.innerText = res.message;
          alert.classList.remove('d-none');
        }
      }).checkLogin(u, p);
    }

    function logout() {
      document.getElementById('dashboardView').classList.add('d-none');
      document.getElementById('loginView').classList.remove('d-none');
      document.getElementById('loginForm').reset();
    }

    // 2. LOAD DASHBOARD DATA
    function loadDashboard() {
      google.script.run.withSuccessHandler(function(data) {
        globalData = data;
        renderKPI(data.kpi);
        renderCharts(data);
        renderTransactions(data.transactions);
      }).getDashboardData();
    }

    // 3. RENDER TOP TIER (KPI)
    function renderKPI(kpiList) {
      if (!kpiList || kpiList.length === 0) return;
      const kpi = kpiList[0];

      document.getElementById('kpiNPV').innerText = 'Rp ' + (kpi.NPV_IDR / 1000000000).toFixed(2) + ' B';
      document.getElementById('kpiIRR').innerText = kpi.IRR_Percent + '%';
      document.getElementById('kpiProfitMargin').innerText = kpi.Profit_Margin_Percent + '%';
      document.getElementById('kpiBudgetVariance').innerText = (kpi.Budget_Variance_Percent > 0 ? '+' : '') + kpi.Budget_Variance_Percent + '%';
      document.getElementById('kpiPayback').innerText = kpi.Payback_Period_Years + ' Tahun';

      // Lampu Lalu Lintas RAG pada Kartu Anggaran
      const cardVar = document.getElementById('cardVariance');
      if (kpi.Budget_Variance_Percent > 5) {
        cardVar.className = 'kpi-card rag-red';
        document.getElementById('kpiVarianceStatus').innerText = 'Bahaya: Overbudget';
      } else if (kpi.Budget_Variance_Percent > 0) {
        cardVar.className = 'kpi-card rag-yellow';
        document.getElementById('kpiVarianceStatus').innerText = 'Perhatian: Mendekati Limit';
      } else {
        cardVar.className = 'kpi-card rag-green';
        document.getElementById('kpiVarianceStatus').innerText = 'Aman: Hemat Anggaran';
      }
    }

    // 4. RENDER MIDDLE TIER (CHARTS)
    function renderCharts(data) {
      // SCurve
      const labels = data.scurve.map(d => d.Bulan);
      const sPlan = data.scurve.map(d => d.Progres_Rencana_Pct);
      const sAct = data.scurve.map(d => d.Progres_Aktual_Pct);

      if (chartSCurveObj) chartSCurveObj.destroy();
      chartSCurveObj = new Chart(document.getElementById('chartSCurve'), {
        type: 'line',
        data: {
          labels: labels,
          datasets: [
            { label: 'Rencana (%)', data: sPlan, borderColor: '#2B6CB0', fill: false },
            { label: 'Aktual (%)', data: sAct, borderColor: '#E53935', fill: false }
          ]
        }
      });

      // EVM
      const evmLabels = data.evm.map(d => d.Bulan);
      const pv = data.evm.map(d => d.PV_IDR / 1000000000);
      const ac = data.evm.map(d => d.AC_IDR / 1000000000);
      const ev = data.evm.map(d => d.EV_IDR / 1000000000);

      if (chartEVMObj) chartEVMObj.destroy();
      chartEVMObj = new Chart(document.getElementById('chartEVM'), {
        type: 'bar',
        data: {
          labels: evmLabels,
          datasets: [
            { label: 'PV (Miliar)', data: pv, backgroundColor: '#1A365D' },
            { label: 'AC (Miliar)', data: ac, backgroundColor: '#E53935' },
            { label: 'EV (Miliar)', data: ev, backgroundColor: '#4CAF50' }
          ]
        }
      });

      // CashFlow
      const cfLabels = data.cashflow.map(d => d.Bulan);
      const cfIn = data.cashflow.map(d => d.Uang_Masuk / 1000000000);
      const cfOut = data.cashflow.map(d => d.Uang_Keluar / 1000000000);
      const cfCum = data.cashflow.map(d => d.Saldo_Kas_Kumulatif / 1000000000);

      if (chartCashFlowObj) chartCashFlowObj.destroy();
      chartCashFlowObj = new Chart(document.getElementById('chartCashFlow'), {
        data: {
          labels: cfLabels,
          datasets: [
            { type: 'bar', label: 'Uang Masuk (B)', data: cfIn, backgroundColor: '#4CAF50' },
            { type: 'bar', label: 'Uang Keluar (B)', data: cfOut, backgroundColor: '#E53935' },
            { type: 'line', label: 'Saldo Kumulatif (B)', data: cfCum, borderColor: '#1A365D', fill: false }
          ]
        }
      });
    }

    // 5. RENDER BOTTOM TIER (TABEL & DRILL DOWN)
    function renderTransactions(list) {
      const tbody = document.getElementById('transactionBody');
      tbody.innerHTML = '';

      list.forEach(t => {
        let ragBadge = '<span class="badge bg-success">Hijau</span>';
        if (t.Status_RAG === 'Kuning') ragBadge = '<span class="badge bg-warning text-dark">Kuning</span>';
        if (t.Status_RAG === 'Merah') ragBadge = '<span class="badge bg-danger">Merah</span>';

        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td class="fw-bold">${t.ID_Trans}</td>
          <td>${t.Tanggal ? new Date(t.Tanggal).toLocaleDateString('id-ID') : '-'}</td>
          <td>${t.Kategori}</td>
          <td>${t.Tahapan_Proyek}</td>
          <td class="fw-bold text-primary" style="cursor:pointer;" onclick="drillDownVendor('${t.Vendor}')">${t.Vendor}</td>
          <td>${t.Deskripsi}</td>
          <td class="fw-bold">Rp ${Number(t.Jumlah).toLocaleString('id-ID')}</td>
          <td>${ragBadge}</td>
          <td>
            <button class="btn btn-sm btn-outline-warning me-1" onclick="editTransaction('${t.ID_Trans}')"><i class="bi bi-pencil"></i></button>
            <button class="btn btn-sm btn-outline-danger" onclick="deleteTransaction('${t.ID_Trans}')"><i class="bi bi-trash"></i></button>
          </td>
        `;
        tbody.appendChild(tr);
      });
    }

    // 6. FILTER DINAMIS
    function applyFilters() {
      const cat = document.getElementById('filterKategori').value;
      const stage = document.getElementById('filterTahapan').value;

      let filtered = globalData.transactions;
      if (cat !== 'ALL') filtered = filtered.filter(t => t.Kategori === cat);
      if (stage !== 'ALL') filtered = filtered.filter(t => t.Tahapan_Proyek === stage);

      renderTransactions(filtered);
    }

    function resetFilters() {
      document.getElementById('filterPeriode').value = 'ALL';
      document.getElementById('filterKategori').value = 'ALL';
      document.getElementById('filterTahapan').value = 'ALL';
      renderTransactions(globalData.transactions);
    }

    // 7. FITUR DRILL-DOWN VENDOR
    function drillDownVendor(vendorName) {
      alert(`[Drill-Down Investigasi]\nVendor: ${vendorName}\nSistem menyaring detail transaksi terkait vendor ini.`);
      const filtered = globalData.transactions.filter(t => t.Vendor === vendorName);
      renderTransactions(filtered);
    }

    // 8. HANDLER CRUD FORM
    function openModal() {
      document.getElementById('transForm').reset();
      document.getElementById('formID').value = '';
      document.getElementById('modalTitle').innerText = 'Tambah Transaksi Baru';
      modalObj.show();
    }

    function editTransaction(id) {
      const item = globalData.transactions.find(t => t.ID_Trans === id);
      if (!item) return;

      document.getElementById('formID').value = item.ID_Trans;
      document.getElementById('formTanggal').value = item.Tanggal ? new Date(item.Tanggal).toISOString().split('T')[0] : '';
      document.getElementById('formKategori').value = item.Kategori;
      document.getElementById('formTahapan').value = item.Tahapan_Proyek;
      document.getElementById('formVendor').value = item.Vendor;
      document.getElementById('formDeskripsi').value = item.Deskripsi;
      document.getElementById('formJumlah').value = item.Jumlah;
      document.getElementById('formRAG').value = item.Status_RAG;

      document.getElementById('modalTitle').innerText = 'Edit Transaksi (' + id + ')';
      modalObj.show();
    }

    function handleSaveTransaction(e) {
      e.preventDefault();
      const payload = {
        ID_Trans: document.getElementById('formID').value,
        Tanggal: document.getElementById('formTanggal').value,
        Kategori: document.getElementById('formKategori').value,
        Tahapan_Proyek: document.getElementById('formTahapan').value,
        Vendor: document.getElementById('formVendor').value,
        Deskripsi: document.getElementById('formDeskripsi').value,
        Jumlah: document.getElementById('formJumlah').value,
        Status_RAG: document.getElementById('formRAG').value
      };

      google.script.run.withSuccessHandler(function(res) {
        alert(res.message);
        modalObj.hide();
        loadDashboard();
      }).saveTransaction(payload);
    }

    function deleteTransaction(id) {
      if (confirm('Apakah Anda yakin ingin menghapus transaksi ' + id + '?')) {
        google.script.run.withSuccessHandler(function(res) {
          alert(res.message);
          loadDashboard();
        }).deleteTransaction(id);
      }
    }
  </script>
